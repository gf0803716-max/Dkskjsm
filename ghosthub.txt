local moving = false
local function killPart()
    local c = player.Character
    if c and c:FindFirstChild("PartTele") then c.PartTele:Destroy() end
end

local function makePart()
    local c = player.Character
    if not c then return nil end
    local p = c:FindFirstChild("PartTele")
    if p then return p end
    p = Instance.new("Part")
    p.Size = Vector3.new(10, 1, 10)
    p.Name = "PartTele"
    p.Anchored = true
    p.Transparency = 1
    p.CanCollide = true
    p.CFrame = c.HumanoidRootPart.CFrame
    p.Parent = c
    p:GetPropertyChangedSignal("CFrame"):Connect(function()
        if moving then
            local ch = player.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if hrp then hrp.CFrame = p.CFrame end
        end
    end)
    return p
end

-- viaja até um CFrame. valid() = continua? tpRange > 0 ativa Tween+TP no final
local function travel(target, valid, speed, tpRange, boost)
    if not alive() then return false end
    local part = makePart()
    if not part then return false end
    moving = true

    while moving and (not valid or valid()) do
        local hrp = root()
        if not hrp then break end
        local dist = (target.Position - hrp.Position).Magnitude
        if dist <= S.ReachDist then break end

        -- Tween + TP: nos ultimos metros dá TP direto (igual TP puro)
        if tpRange and tpRange > 0 and dist <= tpRange then
            part.CFrame = target
            hrp.CFrame = target
            break
        end

        local sp = speed
        if boost and S.BoostOn and dist <= S.BoostDist then sp = S.BoostSpeed end
        sp = math.clamp(sp, 1, 10000)

        local step = math.min(dist, 900)
        local goal = CFrame.new(hrp.Position + (target.Position - hrp.Position).Unit * step)
        if step >= dist then goal = target end

        local tw = TweenService:Create(part, TweenInfo.new(step / sp, Enum.EasingStyle.Linear), { CFrame = goal })
        tw:Play()
        local done = false
        tw.Completed:Once(function() done = true end)
        while not done and moving and (not valid or valid()) do task.wait(0.05) end
        if not done then tw:Cancel() end
    end

    moving = false
    killPart()
    return true
end

local function tpTo(cf)
    local hrp = root()
    if hrp then
        hrp.CFrame = cf
        hrp.AssemblyLinearVelocity = Vector3.zero
    end
end

local boxes = {}
local function chestKind(name)
    local n = name:lower()
    if n:find("chest3") or n:find("diamond") then return "Diamond" end
    if n:find("chest2") or n:find("gold") then return "Gold" end
    return "Silver"
end
local BOXCOLOR = {
    Silver  = Color3.fromRGB(235, 235, 235),
    Gold    = Color3.fromRGB(255, 205, 0),
    Diamond = Color3.fromRGB(0, 120, 255),
}

local function dressChest(chest)
    if not chest or not chest.Parent then return end
    if S.NoSkin or S.BoxChest then
        for _, d in ipairs(chest:GetDescendants()) do
            if d:IsA("BasePart") then
                pcall(function() d.Transparency = 1 end)
            elseif d:IsA("Animator") or d:IsA("ParticleEmitter") or d:IsA("Beam") or d:IsA("Trail") then
                pcall(function() d.Enabled = false end)
            elseif d:IsA("AnimationTrack") then
                pcall(function() d:Stop() end)
            end
        end
    end
    if S.BoxChest and not boxes[chest] then
        local ok, pivot = pcall(function() return chest:GetPivot() end)
        if not ok then return end
        local b = Instance.new("Part")
        b.Size = Vector3.new(4, 4, 4)
        b.Anchored = true
        b.CanCollide = false
        b.Material = Enum.Material.SmoothPlastic
        b.Color = BOXCOLOR[chestKind(chest.Name)]
        b.CFrame = pivot
        b.Parent = workspace
        boxes[chest] = b
    end
end

local function clearBoxes()
    for k, b in pairs(boxes) do pcall(function() b:Destroy() end); boxes[k] = nil end
end

----------------------------------------------------------------- AUTO CHEST
local function getBeli()
    local d = player:FindFirstChild("Data")
    local b = d and d:FindFirstChild("Beli")
    return (b and b.Value) or 0
end

local statusTxt = "Parado"
local chestCount, beliStart = 0, getBeli()

local function pickChest()
    local hrp = root()
    if not hrp then return nil end
    local pos = hrp.Position
    local best, bd = nil, math.huge
    for _, chest in ipairs(CollectionService:GetTagged("_ChestTagged")) do
        if chest.Parent and not chest:GetAttribute("IsDisabled") then
            local k = chestKind(chest.Name)
            local ok = (k == "Silver" and S.FarmSilver) or (k == "Gold" and S.FarmGold) or (k == "Diamond" and S.FarmDiamond)
            if ok then
                local okp, piv = pcall(function() return chest:GetPivot() end)
                if okp then
                    local d = (piv.Position - pos).Magnitude
                    if d < bd then best, bd = chest, d end
                end
            end
        end
    end
    return best, bd
end

task.spawn(function()
    while true do
        task.wait(0.05)
        if S.FarmChest then
            pcall(function()
                if checkLegend() then return end
                if not alive() then statusTxt = "Respawn..." ; return end

                local chest = pickChest()
                if not chest then statusTxt = "Sem baú" ; return end

                dressChest(chest)
                local piv = chest:GetPivot()
                local target = CFrame.new(piv.Position)
                local beliBefore = getBeli()

                local function valid()
                    return S.FarmChest and chest.Parent and not chest:GetAttribute("IsDisabled")
                end

                if S.Method == "TP" then
                    statusTxt = "TP no baú"
                    tpTo(target)
                    if S.TPDelay > 0 then
                        local t = 0
                        while t < S.TPDelay and valid() do task.wait(0.05); t = t + 0.05 end
                    else
                        task.wait(S.GrabTime)
                    end
                else
                    statusTxt = (S.Method == "TweenTP") and "Tween+TP" or "Tween"
                    travel(target, valid, S.TweenSpeed, (S.Method == "TweenTP") and S.TPRange or 0, true)
                    task.wait(S.GrabTime)   -- Grab & Go: pega e mete o pé
                end

                -- verificacao de dinheiro (opcional, no intervalo escolhido)
                if S.BeliCheck then
                    local t = 0
                    while t < S.BeliEvery do
                        task.wait(0.1); t = t + 0.1
                        if getBeli() > beliBefore then break end
                    end
                end
                if getBeli() > beliBefore then chestCount = chestCount + 1 end
                if boxes[chest] and (not chest.Parent or chest:GetAttribute("IsDisabled")) then
                    pcall(function() boxes[chest]:Destroy() end); boxes[chest] = nil
                end
            end)
        else
            statusTxt = "Parado"
        end
    end
end)

----------------------------------------------------------------- FRUITS
local function fruitRarity(name)
    local n = name:lower()
    for r, list in pairs(RARITY) do
        for _, k in ipairs(list) do if n:find(k) then return r end end
    end
    return "Common"
end
local function fruitBlocked(name)
    local n = name:lower()
    if S.UseBlacklist then
        for _, b in ipairs(BLACKLIST) do if n:find(b) then return true end end
    end
    local r = fruitRarity(name)
    if not S.Rarity[r] then return true end
    return false
end

local function dropped()
    local out = {}
    for _, v in ipairs(workspace:GetChildren()) do
        if v.Name:find("Fruit") or (v:IsA("Tool") and v.Name:find("Fruit")) then
            local part = v:IsA("BasePart") and v or v:FindFirstChildWhichIsA("BasePart", true)
            if part and not fruitBlocked(v.Name) then
                table.insert(out, { obj = v, part = part })
            end
        end
    end
    return out
end

local seen = {}
task.spawn(function()
    while true do
        task.wait(1)
        if S.FruitNotify then
            pcall(function()
                for _, f in ipairs(dropped()) do
                    if not seen[f.obj] then
                        seen[f.obj] = true
                        Notify("FRUTA!", f.obj.Name .. " (" .. fruitRarity(f.obj.Name) .. ")", Color3.fromRGB(0, 200, 255))
                        ChatMsg("[GHOST HUB] Fruta dropada: " .. f.obj.Name, Color3.fromRGB(0, 200, 255))
                    end
                end
            end)
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(0.15)
        if S.AutoFruit then
            pcall(function()
                local hrp = root()
                if not hrp then return end
                local list = dropped()
                if #list == 0 then return end
                table.sort(list, function(a, b)
                    return (a.part.Position - hrp.Position).Magnitude < (b.part.Position - hrp.Position).Magnitude
                end)
                local f = list[1]
                if S.FruitMode == "Bring" then
                    -- a fruta vem pra você
                    if f.part.Anchored then f.part.Anchored = false end
                    f.part.CFrame = hrp.CFrame * CFrame.new(0, 2, -2)
                    f.part.AssemblyLinearVelocity = Vector3.zero
                elseif S.FruitMode == "TP" then
                    tpTo(CFrame.new(f.part.Position + Vector3.new(0, 3, 0)))
                    task.wait(0.2)
                else
                    travel(CFrame.new(f.part.Position + Vector3.new(0, 3, 0)),
                        function() return S.AutoFruit and f.obj.Parent end, S.FruitSpeed, 0, false)
                end
            end)
        end
    end
end)

----------------------------------------------------------------- MOB PULL (modelo inteiro)
local function mobFolders()
    local t = {}
    for _, n in ipairs({ "Enemies", "Mobs", "RaidEnemies", "Dungeon", "DungeonEnemies", "SeaBeasts" }) do
        local f = workspace:FindFirstChild(n)
        if f then table.insert(t, f) end
    end
    return t
end

local function questNames()
    local names = {}
    local gui = player:FindFirstChild("PlayerGui")
    local main = gui and gui:FindFirstChild("Main")
    local q = main and main:FindFirstChild("Quest")
    if q then
        for _, d in ipairs(q:GetDescendants()) do
            if d:IsA("TextLabel") and d.Text ~= "" then table.insert(names, d.Text:lower()) end
        end
    end
    return names
end

local function mobs()
    local hrp = root()
    local out = {}
    if not hrp then return out end
    local qn = S.MobQuestOnly and questNames() or nil
    for _, folder in ipairs(mobFolders()) do
        if not S.MobRaid and (folder.Name:find("Raid") or folder.Name:find("Dungeon")) then
            -- pula
        else
            for _, m in ipairs(folder:GetChildren()) do
                local h = m:FindFirstChildOfClass("Humanoid")
                local r = m:FindFirstChild("HumanoidRootPart")
                if h and r and h.Health > 0 then
                    local d = (r.Position - hrp.Position).Magnitude
                    local ok = d <= S.MobRadius
                    if ok and qn then
                        ok = false
                        for _, nm in ipairs(qn) do
                            if nm:find(m.Name:lower(), 1, true) then ok = true break end
                        end
                    end
                    if ok then table.insert(out, { m = m, r = r, d = d }) end
                end
            end
        end
    end
    table.sort(out, function(a, b) return a.d < b.d end)
    if S.MobLimit > 0 then while #out > S.MobLimit do table.remove(out) end end
    return out
end

local ang = 0
local hb = RunService.Heartbeat:Connect(function(dt)
    if not S.MobPull then return end
    local hrp = root()
    if not hrp then return end
    local list = mobs()
    local total = math.max(#list, 1)
    if S.MobSpin then ang = (ang + dt * S.MobSpinSpeed) % (math.pi * 2) end
    for i, e in ipairs(list) do
        local a = (math.pi * 2 / total) * (i - 1) + ang
        local pos = hrp.Position + Vector3.new(math.cos(a) * S.MobCircle, S.MobHeight, math.sin(a) * S.MobCircle)
        local dest = CFrame.new(pos, hrp.Position)
        -- move o MODELO todo (nao so a skin) e deixa tomar dano
        pcall(function()
            local h = e.m:FindFirstChildOfClass("Humanoid")
            if h then h.PlatformStand = false; h.WalkSpeed = 0 end
            if S.MobMode == "TP" then
                e.m:PivotTo(dest)
            else
                local t = math.clamp(280 / math.max(S.TweenSpeed, 1), 0.05, 1)
                TweenService:Create(e.r, TweenInfo.new(t, Enum.EasingStyle.Linear), { CFrame = dest }):Play()
            end
            e.r.AssemblyLinearVelocity = Vector3.zero
            e.r.AssemblyAngularVelocity = Vector3.zero
        end)
    end
end)

-- widgets
local function glow(frame, on)
    local st = frame:FindFirstChildOfClass("UIStroke")
    if not st then return end
    TweenService:Create(st, TweenInfo.new(0.25), {
        Color = on and NEON or Color3.fromRGB(40, 48, 60),
        Transparency = on and 0 or 0.4,
        Thickness = on and 2 or 1,
    }):Play()
end

local function Toggle(page, label, default, cb)
    local f = Instance.new("Frame", page)
    f.Size = UDim2.new(1, -6, 0, 34)
    f.BackgroundColor3 = BG2
    f.BorderSizePixel = 0
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)
    local st = Instance.new("UIStroke", f)
    st.Color = Color3.fromRGB(40, 48, 60); st.Thickness = 1; st.Transparency = 0.4

    local dot = Instance.new("Frame", f)
    dot.Size = UDim2.fromOffset(8, 8)
    dot.Position = UDim2.fromOffset(10, 13)
    dot.BackgroundColor3 = Color3.fromRGB(70, 80, 95)
    dot.BorderSizePixel = 0
    Instance.new("UICorner", dot).CornerRadius = UDim.new(1, 0)

    local t = Instance.new("TextLabel", f)
    t.BackgroundTransparency = 1
    t.Position = UDim2.fromOffset(26, 0)
    t.Size = UDim2.new(1, -34, 1, 0)
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.Font = Enum.Font.GothamMedium
    t.TextSize = 12
    t.TextColor3 = WHITE
    t.Text = label

    local btn = Instance.new("TextButton", f)
    btn.BackgroundTransparency = 1
    btn.Size = UDim2.new(1, 0, 1, 0)
    btn.Text = ""

    local on = default
    local function apply(anim)
        glow(f, on)
        TweenService:Create(dot, TweenInfo.new(0.2), { BackgroundColor3 = on and NEON or Color3.fromRGB(70, 80, 95) }):Play()
        if anim then
            if on then Notify(label, "ATIVADO", NEON) else Notify(label, "Desativado", Color3.fromRGB(255, 120, 120)) end
        end
    end
    btn.MouseButton1Click:Connect(function()
        on = not on
        apply(true)
        cb(on)
    end)
    apply(false)
    return f
end

local function Box(page, label, default, cb)
    local f = Instance.new("Frame", page)
    f.Size = UDim2.new(1, -6, 0, 34)
    f.BackgroundColor3 = BG2
    f.BorderSizePixel = 0
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)
    local st = Instance.new("UIStroke", f)
    st.Color = Color3.fromRGB(40, 48, 60); st.Thickness = 1; st.Transparency = 0.4

    local t = Instance.new("TextLabel", f)
    t.BackgroundTransparency = 1
    t.Position = UDim2.fromOffset(10, 0)
    t.Size = UDim2.new(0.58, 0, 1, 0)
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.Font = Enum.Font.GothamMedium
    t.TextSize = 12
    t.TextColor3 = WHITE
    t.Text = label

    local input = Instance.new("TextBox", f)
    input.Size = UDim2.new(0.36, 0, 0, 24)
    input.Position = UDim2.new(0.6, 0, 0, 5)
    input.BackgroundColor3 = BG3
    input.TextColor3 = NEON
    input.Font = Enum.Font.GothamBold
    input.TextSize = 12
    input.ClearTextOnFocus = false
    input.Text = tostring(default)
    Instance.new("UICorner", input).CornerRadius = UDim.new(0, 6)

    input.Focused:Connect(function() glow(f, true) end)
    input.FocusLost:Connect(function()
        local v = tonumber(input.Text)
        if v then cb(v) else input.Text = tostring(default) end
        glow(f, false)
    end)
    return input
end

local function Cycle(page, label, opts, default, cb)
    local f = Instance.new("Frame", page)
    f.Size = UDim2.new(1, -6, 0, 34)
    f.BackgroundColor3 = BG2
    f.BorderSizePixel = 0
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)
    local st = Instance.new("UIStroke", f)
    st.Color = Color3.fromRGB(40, 48, 60); st.Thickness = 1; st.Transparency = 0.4

    local idx = 1
    for i, o in ipairs(opts) do if o == default then idx = i end end

    local btn = Instance.new("TextButton", f)
    btn.BackgroundTransparency = 1
    btn.Size = UDim2.new(1, 0, 1, 0)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 12
    btn.TextColor3 = BLUE
    btn.Text = label .. ": " .. opts[idx]
    btn.MouseButton1Click:Connect(function()
        idx = (idx % #opts) + 1
        btn.Text = label .. ": " .. opts[idx]
        glow(f, true); task.delay(0.5, function() glow(f, false) end)
        cb(opts[idx])
    end)
    return btn
end
